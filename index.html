import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js"; import { getAuth, signInWithEmailAndPassword, createUserWithEmailAndPassword, signOut, onAuthStateChanged, sendPasswordResetEmail, updateProfile } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-auth.js"; import { getFirestore, doc, setDoc, getDoc, collection, getDocs } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; // ⚠️ 請務必替換為您在 Firebase Console 申請的真實 Config 資訊 const firebaseConfig = { apiKey: "YOUR_API_KEY", authDomain: "YOUR_PROJECT_ID.firebaseapp.com", projectId: "YOUR_PROJECT_ID", storageBucket: "YOUR_PROJECT_ID.appspot.com", messagingSenderId: "YOUR_SENDER_ID", appId: "YOUR_APP_ID" }; const app = initializeApp(firebaseConfig); const auth = getAuth(app); const db = getFirestore(app); let currentUser = null; let userPlans = {}; // 儲存該使用者所有的週期計畫 let activePlanId = null; // ========================================== // 1. 雲端資料連動核心邏輯 (Firestore Sync) // ========================================== // 監聽登入狀態切換：換裝置/開啟網頁時自動載入該帳號資料 onAuthStateChanged(auth, async (user) => { if (user) { currentUser = user; console.log("登入成功，使用者 UID:", user.uid); // 更新 UI 顯示使用者名稱/信箱 updateUserUI(user); // 從雲端 Firestore 載入該帳號專屬的資料 await loadUserDataFromCloud(user.uid); // 隱藏登入遮罩，顯示主系統 document.getElementById("authOverlay").classList.add("hidden"); } else { currentUser = null; userPlans = {}; activePlanId = null; console.log("未登入或已登出"); // 顯示登入遮罩 document.getElementById("authOverlay").classList.remove("hidden"); } }); // 從雲端 Firestore 讀取資料 async function loadUserDataFromCloud(uid) { try { const plansRef = collection(db, "users", uid, "plans"); const querySnapshot = await getDocs(plansRef); userPlans = {}; querySnapshot.forEach((docSnap) => { userPlans[docSnap.id] = docSnap.data(); }); // 若雲端尚無資料，建立預設的 2028 年計畫 if (Object.keys(userPlans).length === 0) { const defaultPlan = createNewPlanObject("2028年前必做100件事", "2028-12-31"); userPlans[defaultPlan.id] = defaultPlan; await savePlanToCloud(defaultPlan); } // 預設選取第一個計畫或上次選取的計畫 activePlanId = Object.keys(userPlans)[0]; // 渲染畫面 renderPlanSelector(); renderActivePlan(); } catch (error) { console.error("雲端資料載入失敗:", error); alert("雲端資料載入失敗，請檢查網路連線。"); } } // 儲存單一計畫至雲端 Firestore async function savePlanToCloud(plan) { if (!currentUser) return; try { const planDocRef = doc(db, "users", currentUser.uid, "plans", plan.id); await setDoc(planDocRef, plan, { merge: true }); console.log(`計畫 ${plan.id} 已同步更新至雲端`); } catch (error) { console.error("同步至雲端失敗:", error); } } // ========================================== // 2. 使用者操作事件 (打勾、修改心得、新增週期) // ========================================== // 當使用者勾選/取消勾選目標時 window.toggleGoal = async function(goalId) { const plan = getActivePlan(); if (!plan) return; const goal = plan.goals.find(g => g.id === goalId); if (goal) { goal.completed = !goal.completed; goal.completedAt = goal.completed ? new Date().toISOString() : null; // 重新渲染畫面 renderActivePlan(); // 自動即時連動同步至雲端 await savePlanToCloud(plan); } }; // 當使用者修改年度心得/畫布時 window.saveReflections = async function(field, value) { const plan = getActivePlan(); if (!plan) return; if (!plan.reflections) plan.reflections = {}; plan.reflections[field] = value; // 自動即時連動同步至雲端 await savePlanToCloud(plan); }; // 取得當前作用中的計畫 (防呆與安全取得) function getActivePlan() { if (!activePlanId || !userPlans[activePlanId]) { const keys = Object.keys(userPlans); if (keys.length > 0) { activePlanId = keys[0]; } else { return null; } } return userPlans[activePlanId]; } // ========================================== // 3. 帳號註冊 / 登入 / 忘記密碼處理 // ========================================== // 登入 window.handleLogin = async function(email, password) { try { await signInWithEmailAndPassword(auth, email, password); } catch (error) { alert("登入失敗：" + getErrorMessage(error.code)); } }; // 註冊 window.handleRegister = async function(name, email, password) { try { const res = await createUserWithEmailAndPassword(auth, email, password); if (name) { await updateProfile(res.user, { displayName: name }); } } catch (error) { alert("註冊失敗：" + getErrorMessage(error.code)); } }; // 忘記密碼 window.handleForgotPassword = async function(email) { if (!email) { alert("請先輸入您的電子信箱！"); return; } try { await sendPasswordResetEmail(auth, email); alert("重設密碼信件已發送至您的信箱，請至信箱收信並重設密碼。"); } catch (error) { alert("發送失敗：" + getErrorMessage(error.code)); } }; // 登出 window.handleLogout = async function() { try { await signOut(auth); location.reload(); } catch (error) { console.error("登出失敗", error); } }; // Firebase 錯誤代碼中文轉換 function getErrorMessage(code) { switch (code) { case 'auth/user-not-found': return '找不到此帳號，請先註冊。'; case 'auth/wrong-password': return '密碼輸入錯誤。'; case 'auth/email-already-in-use': return '此信箱已被註冊。'; case 'auth/weak-password': return '密碼強度不足，請輸入至少 6 位數。'; case 'auth/invalid-email': return '請輸入有效的電子信箱格式。'; default: return '發生未知錯誤，請稍後再試。'; } }
